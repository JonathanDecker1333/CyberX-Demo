Python 3

Scripting 
target = "127.0.0.1"
ports = [22, 80, 443]
services = {22: "SSH", 80: "HTTP"}

def audit_target(ip, port_list):
    for p in port_list:
        name = services.get(p, "Unknown")
        print(f"[{ip}] Port {p}: {name}")

audit_target(target, ports)

Go (Golang)

Backend 
package main
import ("fmt"; "net")

func main() {
    addr := "127.0.0.1:80"
    conn, err := net.Dial("tcp", addr)
    if err != nil {
        fmt.Printf("Closed\n")
        return
    }
    conn.Close()
    fmt.Printf("Open!\n")
}
